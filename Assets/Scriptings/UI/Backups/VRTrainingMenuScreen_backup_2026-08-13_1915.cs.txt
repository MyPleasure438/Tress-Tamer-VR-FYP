    private void Start()
    {
        // Sync the displayed/stored floor height to the XR Origin's ACTUAL current Y position on
        // start, so the value shown always reflects reality rather than a stale default.
        if (xrOrigin != null)
            floorHeightCm = xrOrigin.position.y * 100f; // meters -> cm
        if (hideGameplayUntilTrainingStarts)
            SetGameplayActive(false);

        if (buildOnStart)
            BuildScreen();

        if (showMenuOnStart)
            ShowMainMenu();
        else
            SetScreenVisible(false);
    }

    [ContextMenu("Rebuild VR Training Menu")]
    public void RebuildScreen()
    {
        if (screenRoot != null)
        {
            if (Application.isPlaying)
                Destroy(screenRoot);
            else
                DestroyImmediate(screenRoot);
        }

        screenRoot = null;
        canvas = null;
        screenRect = null;
        contentRoot = null;

        BuildScreen();
        ShowMainMenu();
    }

    [ContextMenu("Refresh Current Menu Page")]
    public void RefreshCurrentMenuPage()
    {
        if (screenRoot == null)
            return;

        switch (currentPage)
        {
            case MenuPage.ModuleSelect:
                ShowModuleSelect();
                return;

            case MenuPage.Stage1Select:
                ShowStage1Select();
                return;

            case MenuPage.Settings:
                ShowSettings();
                return;

            case MenuPage.Records:
                ShowRecords();
                return;

            default:
                ShowMainMenu();
                return;
        }
    }

    public void SetAllDarkRectanglesOffset(Vector2 offset)
    {
        allDarkRectanglesOffset = offset;
        RefreshCurrentMenuPage();
    }

    public void BuildScreen()
    {
        if (screenRoot != null)
            return;

        // Required so VR ray/UI clicks work even if the scene did not already contain an EventSystem.
        EnsureEventSystem();

        screenRoot = new GameObject("VR Training Menu Screen");
        screenRoot.transform.SetParent(transform, false);
        screenRoot.transform.localScale = Vector3.one * Mathf.Max(0.0001f, worldScale);

        canvas = screenRoot.AddComponent<Canvas>();
        canvas.renderMode = RenderMode.WorldSpace;
        canvas.worldCamera = ResolveCamera();

        screenRoot.AddComponent<CanvasScaler>().dynamicPixelsPerUnit = 10f;
        screenRoot.AddComponent<GraphicRaycaster>();
        AddOptionalComponent(screenRoot, "UnityEngine.XR.Interaction.Toolkit.UI.TrackedDeviceGraphicRaycaster, Unity.XR.Interaction.Toolkit");

        screenRect = canvas.GetComponent<RectTransform>();
        screenRect.sizeDelta = screenSize;

        contentRoot = screenRect;
        PositionScreen();
    }

    public void BeginTraining()
    {
        if (haircutManager != null)
            haircutManager.SetTrainingMode(selectedTrainingMode);

        if (selectedAllHaircuts && chainManager != null)
        {
            // Chain mode: figure out which stage index we're on from currentPage, start the chain instead
            // of a single hairstyle. Stage index matches HaircutChainManager's stages array (0 = Stage 1).
            int stageIndex = currentPage == MenuPage.Stage2Select ? 1 : 0;
            chainManager.StartChain(stageIndex);

            if (resultPanel != null)
                resultPanel.HideResults();

            SetGameplayActive(true);
            onTrainingModeChanged?.Invoke(selectedTrainingMode);
            onStartTraining?.Invoke();

            if (hideScreenWhenTrainingStarts)
                SetScreenVisible(false);

            return;
        }

        // Original single-haircut flow, unchanged.
        if (haircutManager != null)
            haircutManager.SetActiveHairstyle(selectedHaircut);

        if (resultPanel != null)
            resultPanel.HideResults();

        SetGameplayActive(true);
        onHaircutSelected?.Invoke(selectedHaircut);
        onTrainingModeChanged?.Invoke(selectedTrainingMode);
        onStartTraining?.Invoke();

        if (hideScreenWhenTrainingStarts)
            SetScreenVisible(false);
    }

    public void ShowMenuAgain()
    {
        ShowMainMenu();
    }

    public void RecenterView()
    {
        RecenterScreenToPlayer();
        onRecenterView?.Invoke();
    }

    public void IncreaseFloorHeight()
    {
        SetFloorHeight(floorHeightCm + Mathf.Max(0.1f, floorHeightStepCm));
    }

    public void DecreaseFloorHeight()
    {
        SetFloorHeight(floorHeightCm - Mathf.Max(0.1f, floorHeightStepCm));
    }

    public void SetFloorHeight(float value)
    {
        floorHeightCm = Mathf.Max(0f, value);

        if (xrOrigin != null)
        {
            Vector3 pos = xrOrigin.position;
            pos.y = floorHeightCm / 100f; // cm -> meters
            xrOrigin.position = pos;
        }

        onFloorHeightChanged?.Invoke(floorHeightCm);

        if (currentPage == MenuPage.Settings)
            ShowSettings();
    }

    public void SetLeftHanded(bool value)
    {
        leftHanded = value;
        onLeftHandedChanged?.Invoke(leftHanded);

        if (currentPage == MenuPage.Settings)
            ShowSettings();
    }

    public void SetVibration(bool value)
    {
        vibrationEnabled = value;
        onVibrationChanged?.Invoke(vibrationEnabled);

        if (currentPage == MenuPage.Settings)
            ShowSettings();
    }

    public void ExitApplication()
    {
        onExitRequested?.Invoke();

#if UNITY_EDITOR
        Debug.Log("[VRTrainingMenuScreen] Exit requested. Application.Quit only works in a built game.");
#else
        Application.Quit();
#endif
    }

    private void EvaluateAndRefreshRecords()
    {
        if (evaluationSystem != null)
            evaluationSystem.EvaluateActiveHaircut();

        ShowRecords();
    }

    private void SelectHaircut(HaircutStyle haircutStyle)
    {
        selectedHaircut = haircutStyle;
        selectedAllHaircuts = false;
        onHaircutSelected?.Invoke(selectedHaircut);
        ShowStage1Select();
    }

    private void SelectAllHaircuts()
    {
        selectedAllHaircuts = true;
        RefreshCurrentMenuPage();
    }

    private void PlayClickSound()
    {
        if (UIAudioManager.Instance != null)
            UIAudioManager.Instance.PlayClickSound(transform.position);
    }

    private void SelectTrainingMode(Stage1TrainingMode mode)
    {
        selectedTrainingMode = mode;
        onTrainingModeChanged?.Invoke(selectedTrainingMode);
        ShowStage1Select();
    }

    private void CreateHeader(string title, string subtitle, bool includeBackButton)
    {
        if (includeBackButton)
            CreateButton("<", screenRect, new Vector2(90f, 90f), new Vector2(-620f, 335f), new Color(0f, 0f, 0f, 0f), true, GoBack, 62);

        CreateLabel(title, screenRect, new Vector2(0f, 315f), 66, textWhite, TextAlignmentOptions.Center, new Vector2(980f, 100f));

        if (!string.IsNullOrEmpty(subtitle))
            CreateLabel(subtitle, screenRect, new Vector2(0f, 240f), 38, textMuted, TextAlignmentOptions.Center, new Vector2(760f, 70f));
    }

    private void GoBack()
    {
        switch (currentPage)
        {
            case MenuPage.ModuleSelect:
            case MenuPage.Settings:
            case MenuPage.Records:
                ShowMainMenu();
                return;

            case MenuPage.Stage1Select:
                ShowModuleSelect();
                return;

            default:
                ShowMainMenu();
                return;
        }
    }

    }

    private Vector2 GetDarkRectanglePosition(Vector2 defaultPosition, Vector2 pageOffset)
    {
        return defaultPosition + allDarkRectanglesOffset + pageOffset;
    }

    private Button CreateButton(string label, RectTransform parent, Vector2 size, Vector2 position, Color color, bool interactable, UnityAction action, int fontSize)
    {
        GameObject buttonObject = new GameObject(string.IsNullOrEmpty(label) ? "Button" : label.Replace("\n", " "));
        buttonObject.transform.SetParent(parent, false);

        RectTransform rectTransform = buttonObject.AddComponent<RectTransform>();
        rectTransform.anchorMin = new Vector2(0.5f, 0.5f);
        rectTransform.anchorMax = new Vector2(0.5f, 0.5f);
        rectTransform.pivot = new Vector2(0.5f, 0.5f);
        rectTransform.sizeDelta = size;
        rectTransform.anchoredPosition = position;

        Image image = buttonObject.AddComponent<Image>();
        image.color = color;

        Button button = buttonObject.AddComponent<Button>();
        button.interactable = interactable;

        ColorBlock colors = button.colors;
        colors.normalColor = Color.white;
        colors.highlightedColor = new Color(1.12f, 1.12f, 1.12f, 1f);
        colors.pressedColor = new Color(0.78f, 0.78f, 0.78f, 1f);
        colors.disabledColor = new Color(0.55f, 0.55f, 0.55f, 0.65f);
        button.colors = colors;

        if (action != null)
            button.onClick.AddListener(action);
            button.onClick.AddListener(PlayClickSound);

        if (!string.IsNullOrEmpty(label))
            CreateLabel(label, rectTransform, Vector2.zero, fontSize, interactable ? textWhite : textMuted, TextAlignmentOptions.Center, size - new Vector2(24f, 16f));

        return button;
    }

    private RectTransform CreatePanel(string name, Vector2 size, Vector2 position, Color color)
    {
        GameObject panelObject = new GameObject(name);
        panelObject.transform.SetParent(contentRoot, false);

        RectTransform rectTransform = panelObject.AddComponent<RectTransform>();
        rectTransform.anchorMin = new Vector2(0.5f, 0.5f);
        rectTransform.anchorMax = new Vector2(0.5f, 0.5f);
        rectTransform.pivot = new Vector2(0.5f, 0.5f);
        rectTransform.sizeDelta = size;
        rectTransform.anchoredPosition = position;

        Image image = panelObject.AddComponent<Image>();
        image.color = color;

        return rectTransform;
    }

    private TMP_Text CreateLabel(string text, RectTransform parent, Vector2 position, int fontSize, Color color, TextAlignmentOptions alignment, Vector2 size)
    {
        GameObject labelObject = new GameObject("Text");
        labelObject.transform.SetParent(parent, false);

        RectTransform rectTransform = labelObject.AddComponent<RectTransform>();
        rectTransform.anchorMin = new Vector2(0.5f, 0.5f);
        rectTransform.anchorMax = new Vector2(0.5f, 0.5f);
        rectTransform.pivot = new Vector2(0.5f, 0.5f);
        rectTransform.sizeDelta = size;
        rectTransform.anchoredPosition = position;

        TextMeshProUGUI textComponent = labelObject.AddComponent<TextMeshProUGUI>();
        textComponent.text = text;
        textComponent.color = color;
        textComponent.alignment = alignment;
        textComponent.fontSize = fontSize;
        textComponent.enableAutoSizing = true;
        textComponent.fontSizeMax = fontSize;
        textComponent.fontSizeMin = 14f;
        textComponent.enableWordWrapping = true;
        textComponent.raycastTarget = false;

        return textComponent;
    }

    private void AddBackdrop()
    {
        GameObject backdropObject = new GameObject("Backdrop");
        backdropObject.transform.SetParent(contentRoot, false);

        RectTransform rectTransform = backdropObject.AddComponent<RectTransform>();
        rectTransform.anchorMin = new Vector2(0.5f, 0.5f);
        rectTransform.anchorMax = new Vector2(0.5f, 0.5f);
        rectTransform.pivot = new Vector2(0.5f, 0.5f);
        rectTransform.sizeDelta = ResolvePositiveSize(backdropSize, screenSize);
        rectTransform.anchoredPosition = backdropOffset;

        Image image = backdropObject.AddComponent<Image>();
        image.color = clearBlack;
        image.raycastTarget = false;

        if (!showSoftCenterSurface)
            return;

        RectTransform softPanel = CreatePanel("Soft Center Surface", ResolvePositiveSize(softCenterSurfaceSize, new Vector2(screenSize.x - 80f, screenSize.y - 250f)), softCenterSurfaceOffset, new Color(0.02f, 0.025f, 0.023f, 0.32f));
        Image softPanelImage = softPanel.GetComponent<Image>();

        if (softPanelImage != null)
            softPanelImage.raycastTarget = false;

        softPanel.SetAsLastSibling();
    }

    private Vector2 ResolvePositiveSize(Vector2 requestedSize, Vector2 fallbackSize)
    {
        return new Vector2(
            requestedSize.x > 0f ? requestedSize.x : Mathf.Max(1f, fallbackSize.x),
            requestedSize.y > 0f ? requestedSize.y : Mathf.Max(1f, fallbackSize.y));
    }

    private void ClearContent()
    {
        if (contentRoot == null)
            return;

        for (int i = contentRoot.childCount - 1; i >= 0; i--)
        {
            Transform child = contentRoot.GetChild(i);

            if (Application.isPlaying)
                Destroy(child.gameObject);
            else
                DestroyImmediate(child.gameObject);
        }
    }

    private void EnsureBuilt()
    {
        if (screenRoot == null)
            BuildScreen();
    }

    private void SetScreenVisible(bool isVisible)
    {
        if (screenRoot != null)
            screenRoot.SetActive(isVisible);
    }

    private void SetGameplayActive(bool isActive)
    {
        if (gameplayRootToEnable != null)
            gameplayRootToEnable.SetActive(isActive);

        if (gameplayObjectsToEnable == null)
            return;

        for (int i = 0; i < gameplayObjectsToEnable.Length; i++)
        {
            if (gameplayObjectsToEnable[i] != null)
                gameplayObjectsToEnable[i].SetActive(isActive);
        }
    }

    private void PositionScreen()
    {
        if (screenRoot == null)
            return;

        if (screenAnchor != null)
        {
            screenRoot.transform.SetPositionAndRotation(screenAnchor.position, screenAnchor.rotation);
            return;
        }

        if (placeInFrontOfPlayerOnStart)
        {
            RecenterScreenToPlayer();
            return;
        }

        screenRoot.transform.localPosition = Vector3.zero;
        screenRoot.transform.localRotation = Quaternion.identity;
    }

    private void RecenterScreenToPlayer()
    {
        Transform cameraTransform = ResolveCameraTransform();

        if (cameraTransform == null || screenRoot == null)
            return;

        Vector3 forward = Vector3.ProjectOnPlane(cameraTransform.forward, Vector3.up);

        if (forward.sqrMagnitude <= 0.0001f)
            forward = cameraTransform.forward;

        forward.Normalize();
        Vector3 position = cameraTransform.position + forward * Mathf.Max(0.25f, screenDistanceFromPlayer);
        position.y += screenHeightOffset;
        screenRoot.transform.position = position;
        screenRoot.transform.rotation = Quaternion.LookRotation(forward, Vector3.up);
    }

    private Camera ResolveCamera()
    {
        Transform cameraTransform = ResolveCameraTransform();
        return cameraTransform != null ? cameraTransform.GetComponent<Camera>() : null;
    }

    private Transform ResolveCameraTransform()
    {
        if (playerCamera != null)
            return playerCamera;

        Camera mainCamera = Camera.main;
        return mainCamera != null ? mainCamera.transform : null;
    }

    private void EnsureEventSystem()
    {
        EventSystem existingEventSystem = FindAnyObjectByType<EventSystem>();

        if (existingEventSystem != null)
        {
            EnsureUiInputModule(existingEventSystem.gameObject);
            return;
        }

        GameObject eventSystemObject = new GameObject("EventSystem");
        eventSystemObject.AddComponent<EventSystem>();
        EnsureUiInputModule(eventSystemObject);
    }

    private void EnsureUiInputModule(GameObject eventSystemObject)
    {
        if (AddOptionalComponent(eventSystemObject, "UnityEngine.InputSystem.UI.InputSystemUIInputModule, Unity.InputSystem"))
            return;

        Debug.LogWarning("[VRTrainingMenuScreen] Could not add InputSystemUIInputModule automatically. Add an Input System UI Input Module to your EventSystem if the VR menu cannot be clicked.");
    }

    private bool AddOptionalComponent(GameObject target, string typeName)
    {
        Type type = Type.GetType(typeName);

        if (type == null || target.GetComponent(type) != null)
            return type != null;

        target.AddComponent(type);
        return true;
    }
