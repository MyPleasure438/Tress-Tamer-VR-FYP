using System.Collections.Generic;
using System;
using UnityEngine;

public enum HaircutStyle
{
    Stage1_InductionCut,
    Stage1_ButchCut,
    Stage1_CrewCut,
    Stage2_StandardBob,
    Stage2_LongBob,
    Stage2_ALineBob,
    Stage3_DisconnectedUndercut,
    Stage3_SlickedBackUndercut,
    Stage3_HardPartUndercut,
    Stage4_LowFade,
    Stage4_MidFade,
    Stage4_HighFade,
    Stage5_AsymmetricalBob,
    Stage5_SideSweptPixie,
    Stage5_AvantGardeAngle
}

public enum HaircutTargetLengthMode
{
    OriginalHairWithZoneGradient,
    UniformAllSections,
    CustomPerSection
}

[Serializable]
public class HaircutSectionTargetLength
{
    public HairSectionType Section;
    [Min(0f)] public float TargetLength = 0.025f;
}

[Serializable]
public class HaircutStyleTemplate
{
    public HaircutStyle Style;
    public HaircutTargetLengthMode TargetLengthMode = HaircutTargetLengthMode.OriginalHairWithZoneGradient;
    [Min(0f)] public float UniformTargetLength = 0.025f;
    public HaircutSectionTargetLength[] CustomSectionTargetLengths;
}

// HANDOFF NOTE:
// Active haircut definition manager. Stage1HaircutManager is only a compatibility wrapper for old scene objects.
// This stores per-hairstyle target length templates and controls Practice vs Assessment behavior.
public class HaircutManager : MonoBehaviour
{
    [Header("Active Haircut")]
    [SerializeField] private HaircutStyle activeHairstyle = HaircutStyle.Stage1_ButchCut;
    [SerializeField] private Stage1TrainingMode trainingMode = Stage1TrainingMode.Practice;
    [SerializeField] private Transform hairCardsRoot;
    [SerializeField] private bool applyTargetsOnStart = true;
    [SerializeField] private bool assignSectionsFromNamesBeforeApplyingTargets = true;

    [Header("Perfect Hairstyle On Play")]
    // Debug/demo option: immediately applies the selected target lengths to show the ideal template.
    [SerializeField] private bool showPerfectHaircutOnStart = false;

    [Header("Hair Cap Visual")]
    [SerializeField] private HairCapController hairCapController;
    [SerializeField] private bool applyHairCapLookOnStart = true;
    [SerializeField] private bool applyHairCapLookWhenChangingHaircut = true;

    [Header("Current Target Settings")]
    // BuiltIn = suggested defaults, Uniform = one length for all sections, Custom = per hair section.
    [SerializeField] private HaircutTargetLengthMode targetLengthMode = HaircutTargetLengthMode.OriginalHairWithZoneGradient;
    [SerializeField] private float uniformTargetLength = 0.025f;
    [SerializeField] private HaircutSectionTargetLength[] customSectionTargetLengths = CreateDefaultSectionTargetLengths(0.025f);
    [SerializeField] private float targetTolerance = 0.0025f;

    [Header("Saved Hairstyle Templates")]
    // Each hairstyle remembers its latest inspector values when switching away and back.
    [SerializeField] private HaircutStyleTemplate[] hairstyleTemplates = CreateDefaultHairstyleTemplates();

    [SerializeField, HideInInspector] private HaircutStyle previousActiveHairstyle;
    [SerializeField, HideInInspector] private bool hasInitializedTemplateMemory;

    public HaircutStyle ActiveHairstyle => activeHairstyle;
    public HaircutStyle ActiveHaircut => activeHairstyle;
    public Stage1TrainingMode TrainingMode => trainingMode;
    // Practice mode is a hard limiter. Assessment mode lets the player overcut and scores the result later.
    public bool ShouldLimitCutsToTarget => trainingMode == Stage1TrainingMode.Practice;
    public float TargetTolerance => Mathf.Max(0f, targetTolerance);

    private void OnValidate()
    {
        EnsureTemplateMemory();
    }

    private void Start()
    {
        EnsureTemplateMemory();

        if (applyTargetsOnStart)
            ApplyTargetsToHairCards();

        // Sync stubble reveal style with whatever hairstyle is already active at Start (e.g. set via the
        // Inspector dropdown rather than through SetActiveHairstyle()), so the two systems never disagree.
        // Deferred one frame: Unity does not guarantee Start() order between scripts, so this avoids a race
        // where StubbleRevealCoordinator's own Start() (which sets its style from Starting Style) could run
        // AFTER this and silently overwrite the sync.
        StartCoroutine(SyncStubbleRevealStyleNextFrame());

        if (showPerfectHaircutOnStart)
            ApplyPerfectHaircutToSceneHair();

        if (applyHairCapLookOnStart)
            ApplyActiveHaircutToHairCap();
    }

    [Tooltip("Optional. If assigned, selecting a Stage 1 style (Induction/Butch/Crew) here also switches which stubble mesh is revealed as hair gets cut - keeps StubbleRevealCoordinator in sync with whatever haircut is actually selected. Leave empty if not using stubble reveal.")]
    [SerializeField] private StubbleRevealCoordinator stubbleRevealCoordinator;

    public void SetActiveHairstyle(HaircutStyle hairstyle)
    {
        EnsureTemplateMemory();
        // Save current inspector settings first so switching hairstyles does not lose custom section lengths.
        SaveCurrentSettingsToTemplate(activeHairstyle);
        activeHairstyle = hairstyle;
        LoadTemplateIntoCurrentSettings(activeHairstyle);
        previousActiveHairstyle = activeHairstyle;
        ApplyTargetsToHairCards();
        SyncStubbleRevealStyle(hairstyle);

        if (applyHairCapLookWhenChangingHaircut)
            ApplyActiveHaircutToHairCap();
    }

    private System.Collections.IEnumerator SyncStubbleRevealStyleNextFrame()
    {
        yield return null;
        SyncStubbleRevealStyle(activeHairstyle);
    }

    // Keeps StubbleRevealCoordinator's active stubble mesh in sync with the Stage 1 style just selected.
    // Does nothing for Stage 2+ styles (bobs etc.) since those do not have a stubble equivalent.
    private void SyncStubbleRevealStyle(HaircutStyle hairstyle)
    {
        if (stubbleRevealCoordinator == null)
            return;

        switch (hairstyle)
        {
            case HaircutStyle.Stage1_InductionCut:
                stubbleRevealCoordinator.SetActiveStyle(StubbleRevealCoordinator.StubbleStyle.Induction);
                break;
            case HaircutStyle.Stage1_ButchCut:
                stubbleRevealCoordinator.SetActiveStyle(StubbleRevealCoordinator.StubbleStyle.Butch);
                break;
            case HaircutStyle.Stage1_CrewCut:
                stubbleRevealCoordinator.SetActiveStyle(StubbleRevealCoordinator.StubbleStyle.Crew);
                break;
        }
    }

    public void SetTrainingMode(Stage1TrainingMode mode)
    {
        trainingMode = mode;
    }

    [ContextMenu("Use Practice Mode")]
    public void UsePracticeMode()
    {
        SetTrainingMode(Stage1TrainingMode.Practice);
    }

    [ContextMenu("Use Assessment Mode")]
    public void UseAssessmentMode()
    {
        SetTrainingMode(Stage1TrainingMode.Assessment);
    }

    [ContextMenu("Use Stage 1 Induction Cut")]
    public void UseStage1InductionCut()
    {
        SetActiveHairstyle(HaircutStyle.Stage1_InductionCut);
    }

    [ContextMenu("Use Stage 1 Butch Cut")]
    public void UseStage1ButchCut()
    {
        SetActiveHairstyle(HaircutStyle.Stage1_ButchCut);
    }

    [ContextMenu("Use Stage 1 Crew Cut")]
    public void UseStage1CrewCut()
    {
        SetActiveHairstyle(HaircutStyle.Stage1_CrewCut);
    }

    [ContextMenu("Apply Targets To Hair Cards")]
    public void ApplyTargetsToHairCards()
    {
        SaveCurrentSettingsToTemplate(activeHairstyle);

        if (assignSectionsFromNamesBeforeApplyingTargets)
            AssignSectionsFromObjectOrParentNames();

        Transform root = hairCardsRoot != null ? hairCardsRoot : transform;
        HairCardData[] hairCards = root.GetComponentsInChildren<HairCardData>(true);

        for (int i = 0; i < hairCards.Length; i++)
        {
            HairCardData hairCard = hairCards[i];

            if (hairCard == null || !TryGetTargetLength(hairCard, out float targetLength, out float tolerance))
                continue;

            hairCard.SetTargetLength(targetLength, tolerance);
        }

        Debug.Log($"Applied {activeHairstyle} target lengths to {hairCards.Length} hair cards.");
    }

    [ContextMenu("Restore Original Hair Length (Undo All Cuts)")]
    public void RestoreOriginalHairLength()
    {
        // Safety tool: SegmentedHairCard.SetVisibleLength never destroys geometry, it only toggles
        // segment visibility - so this is always fully recoverable, no matter how much has been cut.
        Transform root = hairCardsRoot != null ? hairCardsRoot : transform;
        HairCardData[] hairCards = root.GetComponentsInChildren<HairCardData>(true);
        int restoredCount = 0;

        for (int i = 0; i < hairCards.Length; i++)
        {
            HairCardData hairCard = hairCards[i];

            if (hairCard == null || hairCard.OriginalLength <= 0f)
                continue;

            SegmentedHairCard segmentedHairCard = hairCard.GetComponent<SegmentedHairCard>();
            if (segmentedHairCard != null)
                segmentedHairCard.SetVisibleLength(hairCard.OriginalLength);

            hairCard.CurrentLength = hairCard.OriginalLength;
            restoredCount++;
        }

        Debug.Log($"Restored {restoredCount}/{hairCards.Length} hair cards to their original (pre-cut) length.");
    }

    [ContextMenu("Apply Perfect Haircut To Scene Hair")]
    public void ApplyPerfectHaircutToSceneHair()
    {
        // Preview helper only: this modifies visible hair lengths to the current target template.
        if (applyTargetsOnStart)
            ApplyTargetsToHairCards();
        else if (assignSectionsFromNamesBeforeApplyingTargets)
            AssignSectionsFromObjectOrParentNames();

        Transform root = hairCardsRoot != null ? hairCardsRoot : transform;
        HairCardData[] hairCards = root.GetComponentsInChildren<HairCardData>(true);
        int changedCount = 0;

        for (int i = 0; i < hairCards.Length; i++)
        {
            HairCardData hairCard = hairCards[i];

            if (hairCard == null || !TryGetTargetLength(hairCard, out float targetLength, out float tolerance))
                continue;

            SegmentedHairCard segmentedHairCard = hairCard.GetComponent<SegmentedHairCard>();

            if (segmentedHairCard != null)
            {
                segmentedHairCard.SetVisibleLength(Mathf.Max(0f, targetLength));
                changedCount++;
                continue;
            }

            hairCard.CurrentLength = Mathf.Max(0f, targetLength) + Mathf.Max(0f, tolerance);
            changedCount++;
        }

        Debug.Log($"Applied perfect {activeHairstyle} haircut to {changedCount}/{hairCards.Length} hair cards.");
    }

    [ContextMenu("Assign Sections From Object Or Parent Names")]
    public void AssignSectionsFromObjectOrParentNames()
    {
        Transform root = hairCardsRoot != null ? hairCardsRoot : transform;
        HairCardData[] hairCards = root.GetComponentsInChildren<HairCardData>(true);
        int assignedCount = 0;

        for (int i = 0; i < hairCards.Length; i++)
        {
            HairCardData hairCard = hairCards[i];

            if (hairCard == null || !TryFindSectionInHierarchy(hairCard.transform, root, out HairSectionType section))
                continue;

            hairCard.Section = section;
            assignedCount++;
        }

        Debug.Log($"Assigned hair sections from names for {assignedCount}/{hairCards.Length} hair cards.");
    }

    public bool TryGetTargetLength(HairSectionType section, out float targetLength, out float tolerance)
    {
        tolerance = TargetTolerance;

        if (targetLengthMode == HaircutTargetLengthMode.UniformAllSections)
        {
            targetLength = Mathf.Max(0f, uniformTargetLength);
            return true;
        }

        if (targetLengthMode == HaircutTargetLengthMode.CustomPerSection && TryGetCustomSectionTargetLength(section, out targetLength))
            return true;

        return TryGetBuiltInTargetLength(activeHairstyle, section, out targetLength);
    }

    [Header("Within-Zone Length Gradient (bottom-to-top, currently Bob styles only)")]
    [Tooltip("When on, cards near the bottom of a zone get a slightly shorter target and cards near the top get a slightly longer target, instead of one flat length for the whole zone - matches how real bottom-to-top panel cutting naturally leaves a gradient. Only applies to Stage2_StandardBob and Stage2_LongBob for now (the styles that have been tuned) - every other style is completely unaffected.")]
    [SerializeField] private bool enableWithinZoneGradient = true;
    [Tooltip("How strong the bottom-to-top spread is, as a fraction of the zone's normal flat target. 0 = no gradient (identical to old flat behavior). 0.4 = bottom is ~20% shorter and top is ~20% longer than the flat value. Tune by eye in Play mode.")]
    [Range(0f, 1f)]
    [SerializeField] private float gradientRangeFraction = 0.4f;

    private readonly Dictionary<HairSectionType, Vector2> _zoneHeightRangeCache = new Dictionary<HairSectionType, Vector2>();

    // Per-card overload: gradient-aware for eligible styles (currently the tuned Bob styles), falls back
    // to the exact same flat per-zone behavior as the section-only overload above for everything else.
    // Mirrors that method's priority order exactly (Uniform -> CustomPerSection -> built-in) so behavior
    // stays identical whenever gradient doesn't apply.
    public bool TryGetTargetLength(HairCardData hairCard, out float targetLength, out float tolerance)
    {
        tolerance = TargetTolerance;
        targetLength = 0f;

        if (hairCard == null)
            return false;

        if (targetLengthMode == HaircutTargetLengthMode.UniformAllSections)
        {
            targetLength = Mathf.Max(0f, uniformTargetLength);
            return true;
        }

        // In Custom Per Section mode, use the cached custom value as the gradient's CENTER instead of
        // bypassing gradient entirely - this way the bottom-to-top spread is built around whatever value
        // was specifically tuned for that zone (e.g. a manually patched Fringe length), not ignored.
        float? customCenterValue = null;
        if (targetLengthMode == HaircutTargetLengthMode.CustomPerSection && TryGetCustomSectionTargetLength(hairCard.Section, out float customValue))
        {
            customCenterValue = customValue;
        }

        if (enableWithinZoneGradient && IsGradientEligibleStyle(activeHairstyle) && TryGetGradientTargetLength(hairCard, customCenterValue, out float gradientTarget))
        {
            targetLength = gradientTarget;
            return true;
        }

        if (customCenterValue.HasValue)
        {
            targetLength = customCenterValue.Value;
            return true;
        }

        return TryGetBuiltInTargetLength(activeHairstyle, hairCard.Section, out targetLength);
    }

    private static bool IsGradientEligibleStyle(HaircutStyle style)
    {
        // Scoped intentionally to only the styles that have actually been tuned/confirmed so far - every
        // other style keeps its exact original flat behavior. Add more cases here as other styles get tuned.
        return style == HaircutStyle.Stage2_StandardBob || style == HaircutStyle.Stage2_LongBob;
    }

    private bool TryGetGradientTargetLength(HairCardData hairCard, float? centerOverride, out float targetLength)
    {
        targetLength = 0f;

        // Center around a custom-tuned value (Custom Per Section mode) if one was supplied, otherwise the
        // usual code-defined default for this style/zone (Built In Haircut Defaults mode).
        float flatValue = centerOverride ?? GetSuggestedTargetLength(activeHairstyle, hairCard.Section);
        if (flatValue <= 0f)
            return false;

        if (!TryGetRelativeHeightInZone(hairCard, out float t))
        {
            // Can't compute a meaningful gradient (e.g. only one card in this zone) - fall back to the
            // exact flat value rather than guessing, so this never produces a worse result than before.
            targetLength = flatValue;
            return true;
        }

        float halfRange = gradientRangeFraction * 0.5f;
        float bottomValue = flatValue * (1f - halfRange);
        float topValue = flatValue * (1f + halfRange);

        targetLength = Mathf.Lerp(bottomValue, topValue, t);
        return true;
    }

    // Returns this card's relative height (0 = lowest card in its zone, 1 = highest) using world-space Y
    // as an approximation of 'bottom of zone' to 'top of zone'. Cached per zone (lazy, cleared whenever the
    // hairstyle changes) since recomputing min/max across every card in a zone per-call would be wasteful.
    private bool TryGetRelativeHeightInZone(HairCardData hairCard, out float t)
    {
        t = 0f;

        if (!_zoneHeightRangeCache.TryGetValue(hairCard.Section, out Vector2 range))
        {
            Transform root = hairCardsRoot != null ? hairCardsRoot : transform;
            HairCardData[] allCards = root.GetComponentsInChildren<HairCardData>(true);

            float min = float.PositiveInfinity;
            float max = float.NegativeInfinity;
            int countInZone = 0;

            for (int i = 0; i < allCards.Length; i++)
            {
                if (allCards[i] == null || allCards[i].Section != hairCard.Section)
                    continue;

                float y = allCards[i].RootPosition.y;
                if (y < min) min = y;
                if (y > max) max = y;
                countInZone++;
            }

            range = countInZone > 1 ? new Vector2(min, max) : new Vector2(0f, 0f);
            _zoneHeightRangeCache[hairCard.Section] = range;
        }

        if (Mathf.Approximately(range.x, range.y))
            return false;

        t = Mathf.InverseLerp(range.x, range.y, hairCard.RootPosition.y);
        return true;
    }

    [ContextMenu("Reset Current Custom Targets From Active Hairstyle Defaults")]
    public void ResetCurrentCustomTargetsFromActiveHairstyleDefaults()
    {
        HairSectionType[] sections = GetDetailedHairSections();
        customSectionTargetLengths = new HaircutSectionTargetLength[sections.Length];

        for (int i = 0; i < sections.Length; i++)
        {
            TryGetBuiltInTargetLength(activeHairstyle, sections[i], out float targetLength);
            customSectionTargetLengths[i] = new HaircutSectionTargetLength
            {
                Section = sections[i],
                TargetLength = targetLength
            };
        }

        SaveCurrentSettingsToTemplate(activeHairstyle);
    }

    [ContextMenu("Set Current Custom Targets To Uniform Length")]
    public void SetCurrentCustomTargetsToUniformLength()
    {
        customSectionTargetLengths = CreateDefaultSectionTargetLengths(uniformTargetLength);
        SaveCurrentSettingsToTemplate(activeHairstyle);
    }

    [ContextMenu("Rebuild Missing Hairstyle Templates")]
    public void RebuildMissingHairstyleTemplates()
    {
        EnsureTemplatesExist();
    }

    [ContextMenu("Reset All Hairstyle Templates To Suggested Section Lengths")]
    public void ResetAllHairstyleTemplatesToSuggestedSectionLengths()
    {
        hairstyleTemplates = CreateDefaultHairstyleTemplates();
        LoadTemplateIntoCurrentSettings(activeHairstyle);
        previousActiveHairstyle = activeHairstyle;
        hasInitializedTemplateMemory = true;
        ApplyTargetsToHairCards();
        ApplyActiveHaircutToHairCap();
    }

    [ContextMenu("Apply Active Haircut To Hair Cap")]
    public void ApplyActiveHaircutToHairCap()
    {
        if (hairCapController == null)
            return;

        GetHairCapLook(activeHairstyle, out float density, out float darkness, out float opacity);
        hairCapController.SetBuzzLook(density, darkness, opacity);
    }

    private void GetHairCapLook(HaircutStyle style, out float density, out float darkness, out float opacity)
    {
        switch (style)
        {
            case HaircutStyle.Stage1_InductionCut:
                density = 0.85f;
                darkness = 1.45f;
                opacity = 1f;
                return;

            case HaircutStyle.Stage1_ButchCut:
                density = 0.62f;
                darkness = 1.15f;
                opacity = 0.9f;
                return;

            case HaircutStyle.Stage1_CrewCut:
                density = 0.48f;
                darkness = 1.0f;
                opacity = 0.75f;
                return;

            case HaircutStyle.Stage3_DisconnectedUndercut:
            case HaircutStyle.Stage3_SlickedBackUndercut:
            case HaircutStyle.Stage3_HardPartUndercut:
                density = 0.7f;
                darkness = 1.25f;
                opacity = 0.9f;
                return;

            case HaircutStyle.Stage4_LowFade:
                density = 0.55f;
                darkness = 1.05f;
                opacity = 0.75f;
                return;

            case HaircutStyle.Stage4_MidFade:
                density = 0.65f;
                darkness = 1.2f;
                opacity = 0.85f;
                return;

            case HaircutStyle.Stage4_HighFade:
                density = 0.75f;
                darkness = 1.35f;
                opacity = 0.95f;
                return;

            case HaircutStyle.Stage5_SideSweptPixie:
            case HaircutStyle.Stage5_AvantGardeAngle:
                density = 0.35f;
                darkness = 0.9f;
                opacity = 0.55f;
                return;

            default:
                density = 0f;
                darkness = 0f;
                opacity = 0f;
                return;
        }
    }

    private void EnsureTemplateMemory()
    {
        EnsureTemplatesExist();

        if (customSectionTargetLengths == null || customSectionTargetLengths.Length == 0)
            customSectionTargetLengths = CreateDefaultSectionTargetLengths(uniformTargetLength);

        if (!hasInitializedTemplateMemory)
        {
            SaveCurrentSettingsToTemplate(activeHairstyle);
            previousActiveHairstyle = activeHairstyle;
            hasInitializedTemplateMemory = true;
            return;
        }

        if (previousActiveHairstyle != activeHairstyle)
        {
            SaveCurrentSettingsToTemplate(previousActiveHairstyle);
            LoadTemplateIntoCurrentSettings(activeHairstyle);
            previousActiveHairstyle = activeHairstyle;
            return;
        }

        SaveCurrentSettingsToTemplate(activeHairstyle);
    }

    private void EnsureTemplatesExist()
    {
        HaircutStyle[] styles = (HaircutStyle[])Enum.GetValues(typeof(HaircutStyle));
        HaircutStyleTemplate[] rebuiltTemplates = new HaircutStyleTemplate[styles.Length];

        for (int i = 0; i < styles.Length; i++)
        {
            HaircutStyleTemplate existingTemplate = FindTemplate(styles[i]);
            rebuiltTemplates[i] = existingTemplate != null ? existingTemplate : CreateDefaultTemplate(styles[i]);
        }

        hairstyleTemplates = rebuiltTemplates;
    }

    private HaircutStyleTemplate FindTemplate(HaircutStyle style)
    {
        if (hairstyleTemplates == null)
            return null;

        for (int i = 0; i < hairstyleTemplates.Length; i++)
        {
            if (hairstyleTemplates[i] != null && hairstyleTemplates[i].Style == style)
                return hairstyleTemplates[i];
        }

        return null;
    }

    private void SaveCurrentSettingsToTemplate(HaircutStyle style)
    {
        HaircutStyleTemplate template = FindTemplate(style);

        if (template == null)
            return;

        template.TargetLengthMode = targetLengthMode;
        template.UniformTargetLength = Mathf.Max(0f, uniformTargetLength);
        template.CustomSectionTargetLengths = CloneSectionTargets(customSectionTargetLengths);
    }

    private void LoadTemplateIntoCurrentSettings(HaircutStyle style)
    {
        HaircutStyleTemplate template = FindTemplate(style);

        if (template == null)
            template = CreateDefaultTemplate(style);

        targetLengthMode = template.TargetLengthMode;
        uniformTargetLength = Mathf.Max(0f, template.UniformTargetLength);
        customSectionTargetLengths = CloneSectionTargets(template.CustomSectionTargetLengths);

        if (customSectionTargetLengths == null || customSectionTargetLengths.Length == 0)
            customSectionTargetLengths = CreateDefaultSectionTargetLengths(uniformTargetLength);
    }

    private bool TryGetCustomSectionTargetLength(HairSectionType section, out float targetLength)
    {
        if (customSectionTargetLengths != null)
        {
            for (int i = 0; i < customSectionTargetLengths.Length; i++)
            {
                HaircutSectionTargetLength sectionTarget = customSectionTargetLengths[i];

                if (sectionTarget == null || sectionTarget.Section != section)
                    continue;

                targetLength = Mathf.Max(0f, sectionTarget.TargetLength);
                return true;
            }
        }

        return TryGetBuiltInTargetLength(activeHairstyle, section, out targetLength);
    }

    private bool TryGetBuiltInTargetLength(HaircutStyle style, HairSectionType section, out float targetLength)
    {
        targetLength = GetSuggestedTargetLength(style, section);
        return true;
    }

    private static float GetSuggestedTargetLength(HaircutStyle style, HairSectionType section)
    {
        switch (style)
        {
            case HaircutStyle.Stage1_InductionCut:
                return 0.0075f;

            case HaircutStyle.Stage1_ButchCut:
                return 0.025f;

            case HaircutStyle.Stage1_CrewCut:
                if (IsTopOrCrownSection(section))
                    return 0.03f;

                if (section == HairSectionType.BackUpper
                    || section == HairSectionType.TransitionBackLeftAtUpperSection
                    || section == HairSectionType.TransitionBackRightAtUpperSection)
                    return 0.018f;

                return 0.01f;

            case HaircutStyle.Stage2_StandardBob:
                return GetBobTargetLength(section, 0.13f, 0.135f, 0.14f, 0.15f, 0.145f, 0.10f);

            case HaircutStyle.Stage2_LongBob:
                return GetBobTargetLength(section, 0.12f, 0.125f, 0.16f, 0.24f, 0.18f, 0.065f);

            case HaircutStyle.Stage2_ALineBob:
                return GetALineBobTargetLength(section);

            case HaircutStyle.Stage3_DisconnectedUndercut:
                return GetDisconnectedUndercutTargetLength(section);

            case HaircutStyle.Stage3_SlickedBackUndercut:
                return GetSlickedBackUndercutTargetLength(section);

            case HaircutStyle.Stage3_HardPartUndercut:
                return GetHardPartUndercutTargetLength(section);

            case HaircutStyle.Stage4_LowFade:
                return GetFadeTargetLength(section, 0.006f, 0.02f, 0.035f, 0.055f);

            case HaircutStyle.Stage4_MidFade:
                return GetFadeTargetLength(section, 0.004f, 0.012f, 0.03f, 0.05f);

            case HaircutStyle.Stage4_HighFade:
                return GetFadeTargetLength(section, 0.003f, 0.008f, 0.02f, 0.045f);

            case HaircutStyle.Stage5_AsymmetricalBob:
                return GetAsymmetricalBobTargetLength(section);

            case HaircutStyle.Stage5_SideSweptPixie:
                return GetSideSweptPixieTargetLength(section);

            case HaircutStyle.Stage5_AvantGardeAngle:
                return GetAvantGardeAngleTargetLength(section);

            default:
                return 0.025f;
        }
    }

    private static float GetBobTargetLength(HairSectionType section, float nape, float backMiddle, float backUpper, float top, float side, float fringe)
    {
        switch (section)
        {
            case HairSectionType.BackNape:
                return nape;

            case HairSectionType.Back:
            case HairSectionType.BackMiddle:
            // TransitionAtMiddleSection's natural hair length matches Back/BackMiddle far more closely
            // than Sideburn/SideUpper - grouping it here instead avoids exceeding its shorter natural length.
            case HairSectionType.TransitionBackLeftAtMiddleSection:
            case HairSectionType.TransitionBackRightAtMiddleSection:
                return backMiddle;

            case HairSectionType.BackUpper:
            case HairSectionType.TransitionBackLeftAtUpperSection:
            case HairSectionType.TransitionBackRightAtUpperSection:
                return backUpper;

            case HairSectionType.LeftSideSideburn:
            case HairSectionType.LeftSideUpper:
            case HairSectionType.RightSideSideburn:
            case HairSectionType.RightSideUpper:
                return side;

            // Fringe hair drapes down over the face, so it needs its own (shorter) length to actually
            // frame the face like a real bob, instead of sharing Crown/Top's longer length and covering it.
            case HairSectionType.FringeLeft:
            case HairSectionType.FringeMiddle:
            case HairSectionType.FringeRight:
                return fringe;

            default:
                return top;
        }
    }

    private static float GetALineBobTargetLength(HairSectionType section)
    {
        if (IsFrontSection(section))
            return 0.34f;

        if (section == HairSectionType.LeftSideUpper
            || section == HairSectionType.RightSideUpper
            || section == HairSectionType.LeftSideSideburn
            || section == HairSectionType.RightSideSideburn)
            return 0.30f;

        if (IsTopOrCrownSection(section))
            return 0.28f;

        if (section == HairSectionType.BackNape)
            return 0.18f;

        return 0.22f;
    }

    private static float GetDisconnectedUndercutTargetLength(HairSectionType section)
    {
        if (IsTopOrCrownSection(section))
            return 0.13f;

        if (section == HairSectionType.BackNape
            || section == HairSectionType.LeftSideSideburn
            || section == HairSectionType.RightSideSideburn)
            return 0.006f;

        if (section == HairSectionType.BackUpper
            || section == HairSectionType.TransitionBackLeftAtUpperSection
            || section == HairSectionType.TransitionBackRightAtUpperSection)
            return 0.025f;

        return 0.012f;
    }

    private static float GetSlickedBackUndercutTargetLength(HairSectionType section)
    {
        if (IsFrontSection(section) || section == HairSectionType.TopMiddle)
            return 0.16f;

        if (IsTopOrCrownSection(section))
            return 0.14f;

        if (section == HairSectionType.BackNape
            || section == HairSectionType.LeftSideSideburn
            || section == HairSectionType.RightSideSideburn)
            return 0.006f;

        return 0.012f;
    }

    private static float GetHardPartUndercutTargetLength(HairSectionType section)
    {
        if (IsRightSideSection(section))
            return 0.006f;

        if (section == HairSectionType.BackNape)
            return 0.008f;

        if (IsTopOrCrownSection(section) || section == HairSectionType.LeftSideUpper)
            return 0.11f;

        return 0.018f;
    }

    private static float GetFadeTargetLength(HairSectionType section, float shortest, float lowerBlend, float upperBlend, float top)
    {
        switch (section)
        {
            case HairSectionType.BackNape:
            case HairSectionType.LeftSideSideburn:
            case HairSectionType.RightSideSideburn:
                return shortest;

            case HairSectionType.BackMiddle:
            case HairSectionType.LeftSideUpper:
            case HairSectionType.RightSideUpper:
            case HairSectionType.TransitionBackLeftAtMiddleSection:
            case HairSectionType.TransitionBackRightAtMiddleSection:
                return lowerBlend;

            case HairSectionType.BackUpper:
            case HairSectionType.TransitionBackLeftAtUpperSection:
            case HairSectionType.TransitionBackRightAtUpperSection:
                return upperBlend;

            default:
                return top;
        }
    }

    private static float GetAsymmetricalBobTargetLength(HairSectionType section)
    {
        if (IsRightSideSection(section))
            return 0.32f;

        if (IsFrontSection(section))
            return 0.26f;

        if (section == HairSectionType.BackNape)
            return 0.18f;

        return 0.22f;
    }

    private static float GetSideSweptPixieTargetLength(HairSectionType section)
    {
        switch (section)
        {
            case HairSectionType.FringeRight:
            case HairSectionType.TopRight:
                return 0.12f;

            case HairSectionType.FringeMiddle:
            case HairSectionType.TopMiddle:
            case HairSectionType.CrownRight:
                return 0.09f;

            case HairSectionType.FringeLeft:
            case HairSectionType.TopLeft:
            case HairSectionType.CrownLeft:
            case HairSectionType.CrownBack:
                return 0.06f;

            case HairSectionType.BackNape:
            case HairSectionType.LeftSideSideburn:
            case HairSectionType.RightSideSideburn:
                return 0.018f;

            default:
                return 0.035f;
        }
    }

    private static float GetAvantGardeAngleTargetLength(HairSectionType section)
    {
        if (section == HairSectionType.FringeRight || section == HairSectionType.RightSideUpper)
            return 0.26f;

        if (section == HairSectionType.FringeLeft || section == HairSectionType.LeftSideUpper)
            return 0.08f;

        if (IsTopOrCrownSection(section))
            return 0.12f;

        if (section == HairSectionType.BackNape)
            return 0.015f;

        return IsRightSideSection(section) ? 0.20f : 0.06f;
    }

    private static bool IsTopOrCrownSection(HairSectionType section)
    {
        switch (section)
        {
            case HairSectionType.Fringe:
            case HairSectionType.Crown:
            case HairSectionType.CrownBack:
            case HairSectionType.CrownLeft:
            case HairSectionType.CrownRight:
            case HairSectionType.FringeLeft:
            case HairSectionType.FringeMiddle:
            case HairSectionType.FringeRight:
            case HairSectionType.TopLeft:
            case HairSectionType.TopMiddle:
            case HairSectionType.TopRight:
                return true;

            default:
                return false;
        }
    }

    private static bool IsFrontSection(HairSectionType section)
    {
        return section == HairSectionType.Fringe
            || section == HairSectionType.FringeLeft
            || section == HairSectionType.FringeMiddle
            || section == HairSectionType.FringeRight;
    }

    private static bool IsRightSideSection(HairSectionType section)
    {
        return section == HairSectionType.RightSide
            || section == HairSectionType.RightSideSideburn
            || section == HairSectionType.RightSideUpper
            || section == HairSectionType.CrownRight
            || section == HairSectionType.FringeRight
            || section == HairSectionType.TopRight
            || section == HairSectionType.TransitionBackRightAtMiddleSection
            || section == HairSectionType.TransitionBackRightAtUpperSection;
    }

    private bool TryFindSectionInHierarchy(Transform hairCardTransform, Transform stopAtRoot, out HairSectionType section)
    {
        section = HairSectionType.Back;
        int bestNameLength = -1;
        HairSectionType bestSection = section;
        Transform current = hairCardTransform;

        while (current != null)
        {
            string normalizedName = NormalizeSectionName(current.name);
            HairSectionType[] sections = (HairSectionType[])Enum.GetValues(typeof(HairSectionType));

            for (int i = 0; i < sections.Length; i++)
            {
                string sectionName = sections[i].ToString();

                if (sectionName.Length <= bestNameLength || !normalizedName.StartsWith(sectionName))
                    continue;

                bestNameLength = sectionName.Length;
                bestSection = sections[i];
            }

            if (current == stopAtRoot)
                break;

            current = current.parent;
        }

        if (bestNameLength < 0)
            return false;

        section = bestSection;
        return true;
    }

    private string NormalizeSectionName(string objectName)
    {
        if (string.IsNullOrEmpty(objectName))
            return string.Empty;

        System.Text.StringBuilder builder = new System.Text.StringBuilder(objectName.Length);

        for (int i = 0; i < objectName.Length; i++)
        {
            char c = objectName[i];

            if (char.IsLetterOrDigit(c))
                builder.Append(c);
        }

        return builder.ToString();
    }

    private static HaircutStyleTemplate CreateDefaultTemplate(HaircutStyle style)
    {
        float uniformLength = GetSuggestedUniformTargetLength(style);

        HaircutStyleTemplate template = new HaircutStyleTemplate
        {
            Style = style,
            TargetLengthMode = HaircutTargetLengthMode.CustomPerSection,
            UniformTargetLength = uniformLength,
            CustomSectionTargetLengths = CreateSuggestedSectionTargetLengths(style)
        };

        return template;
    }

    private static float GetSuggestedUniformTargetLength(HaircutStyle style)
    {
        switch (style)
        {
            case HaircutStyle.Stage1_InductionCut:
                return 0.0075f;

            case HaircutStyle.Stage1_ButchCut:
                return 0.025f;

            case HaircutStyle.Stage1_CrewCut:
                return 0.02f;

            case HaircutStyle.Stage2_StandardBob:
                return 0.20f;

            case HaircutStyle.Stage2_LongBob:
                return 0.30f;

            case HaircutStyle.Stage2_ALineBob:
                return 0.26f;

            case HaircutStyle.Stage3_DisconnectedUndercut:
            case HaircutStyle.Stage3_SlickedBackUndercut:
            case HaircutStyle.Stage3_HardPartUndercut:
                return 0.06f;

            case HaircutStyle.Stage4_LowFade:
                return 0.03f;

            case HaircutStyle.Stage4_MidFade:
                return 0.025f;

            case HaircutStyle.Stage4_HighFade:
                return 0.018f;

            case HaircutStyle.Stage5_AsymmetricalBob:
                return 0.24f;

            case HaircutStyle.Stage5_SideSweptPixie:
                return 0.06f;

            case HaircutStyle.Stage5_AvantGardeAngle:
                return 0.12f;

            default:
                return 0.025f;
        }
    }

    private static HaircutStyleTemplate[] CreateDefaultHairstyleTemplates()
    {
        HaircutStyle[] styles = (HaircutStyle[])Enum.GetValues(typeof(HaircutStyle));
        HaircutStyleTemplate[] templates = new HaircutStyleTemplate[styles.Length];

        for (int i = 0; i < styles.Length; i++)
            templates[i] = CreateDefaultTemplate(styles[i]);

        return templates;
    }

    private static HaircutSectionTargetLength[] CreateDefaultSectionTargetLengths(float targetLength)
    {
        HairSectionType[] sections = GetDetailedHairSections();
        HaircutSectionTargetLength[] targets = new HaircutSectionTargetLength[sections.Length];

        for (int i = 0; i < sections.Length; i++)
        {
            targets[i] = new HaircutSectionTargetLength
            {
                Section = sections[i],
                TargetLength = Mathf.Max(0f, targetLength)
            };
        }

        return targets;
    }

    private static HaircutSectionTargetLength[] CreateSuggestedSectionTargetLengths(HaircutStyle style)
    {
        HairSectionType[] sections = GetDetailedHairSections();
        HaircutSectionTargetLength[] targets = new HaircutSectionTargetLength[sections.Length];

        for (int i = 0; i < sections.Length; i++)
        {
            targets[i] = new HaircutSectionTargetLength
            {
                Section = sections[i],
                TargetLength = Mathf.Max(0f, GetSuggestedTargetLength(style, sections[i]))
            };
        }

        return targets;
    }

    private static HaircutSectionTargetLength[] CloneSectionTargets(HaircutSectionTargetLength[] source)
    {
        if (source == null)
            return null;

        HaircutSectionTargetLength[] clone = new HaircutSectionTargetLength[source.Length];

        for (int i = 0; i < source.Length; i++)
        {
            if (source[i] == null)
                continue;

            clone[i] = new HaircutSectionTargetLength
            {
                Section = source[i].Section,
                TargetLength = source[i].TargetLength
            };
        }

        return clone;
    }

            HairSectionType.FringeLeft,

            HairSectionType.FringeRight,
            HairSectionType.LeftSideSideburn,
            HairSectionType.LeftSideUpper,
            HairSectionType.RightSideSideburn,
            HairSectionType.RightSideUpper,
            HairSectionType.TopLeft,
            HairSectionType.TopMiddle,
            HairSectionType.TopRight,
            HairSectionType.TransitionBackLeftAtMiddleSection,
            HairSectionType.TransitionBackLeftAtUpperSection,
            HairSectionType.TransitionBackRightAtMiddleSection,
            HairSectionType.TransitionBackRightAtUpperSection
        };
    }
