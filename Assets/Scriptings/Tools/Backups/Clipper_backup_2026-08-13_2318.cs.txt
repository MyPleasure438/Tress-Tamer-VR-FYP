using UnityEngine;
using System.Collections;
using UnityEngine.InputSystem;

// HANDOFF NOTE:
// Active electric clipper tool. It cuts repeatedly while on, unlike Scissor which cuts once per snip.
// Desktop testing can use left mouse. VR should call ToggleClipper(), StartClipper(), or StopClipper()
// from XR Grab Interactable events depending on whether one-press toggle or press-and-hold feels better.
public class Clipper : BaseTool
{
    // Active dependencies for the CuttingManager-based hair-card cutting engine.
    [Header("Cutting Engine Links")]
    [Tooltip("Drag the Hair GameObject with the CuttingManager attached here")]
    [SerializeField] private CuttingManager cuttingManager;
    
    [Tooltip("Create an empty GameObject at the clipper's shaving head and drag it here")]
    [SerializeField] private Transform clipperBladeTip;
    
    [Tooltip("Fallback size when no cut area boxes are assigned.")]
    [SerializeField] private float cutRadius = 0.04f;

    [Tooltip("Length of the fallback cutting cylinder along the blade tip's up direction.")]
    [SerializeField] private float cutCylinderLength = 0.05f;

    [Tooltip("Optional. Assign one or more box colliders around the clipper blade for accurate shaving.")]
    [SerializeField] private BoxCollider[] cutAreaBoxes;

    [Tooltip("Optional. If assigned, cuts will be driven by the currently selected guard length instead of the hairstyle's flat per-zone target - matches how a real clipper guard works. Leave empty to fall back to old behavior.")]
    [SerializeField] private ClipperGuardController clipperGuardController;

    [Header("Clipper Engine Settings")]
    [Tooltip("How often does the clipper calculate cuts (in seconds)? Lower = smoother but heavier on CPU.")]
    [SerializeField] private float cutInterval = 0.05f;

    [Tooltip("On for mouse testing. Off when using XR Activate events.")]
    [SerializeField] private bool useLeftMouseButton = true;

    [Tooltip("If on, pressing activate once turns the clipper on/off. If off, call StartClipper on activate and StopClipper on deactivate.")]
    [SerializeField] private bool toggleOnUse = true;

    private bool isClipperOn = false;
    private Coroutine clipperRoutine;

    void Update()
    {
        if (!useLeftMouseButton || Mouse.current == null)
            return;

        if (toggleOnUse)
        {
            if (Mouse.current.leftButton.wasPressedThisFrame)
                ToggleClipper();

            return;
        }

        if (Mouse.current.leftButton.wasPressedThisFrame && !isClipperOn)
            StartClipper();

        if (Mouse.current != null && Mouse.current.leftButton.wasReleasedThisFrame && isClipperOn)
            StopClipper();
    }

    private void OnDisable()
    {
        StopClipper();
    }

    public void ToggleClipper()
    {
        if (isClipperOn)
            StopClipper();
        else
            StartClipper();
    }

    public void StartClipper()
    {
        if (isClipperOn)
            return;

        isClipperOn = true;
        Debug.Log("[Clipper] BZZZZZ! Clipper turned on.");
        clipperRoutine = StartCoroutine(ContinuousShaveRoutine());
    }

    public void StopClipper()
    {
        if (!isClipperOn)
            return;

        isClipperOn = false;
        Debug.Log("[Clipper] Clipper turned off.");

        if (clipperRoutine != null)
        {
            StopCoroutine(clipperRoutine);
            clipperRoutine = null;
        }
    }

    /// <summary>
    /// Loops continuously, firing the cutting calculation at a fixed time interval
    /// </summary>
    IEnumerator ContinuousShaveRoutine()
    {
        while (isClipperOn)
        {
            if (cuttingManager != null && clipperBladeTip != null)
            {
                // Keep runtime work in CuttingManager so scissor and clipper share target-limit/scoring rules.
                playHapticFeedback(cutInterval);
                yield return TriggerClipperCut();
            }
            else
            {
                Debug.LogWarning($"[Clipper] Cut loop ticking but missing references on {gameObject.name}!");
            }

            // Wait for a fraction of a second before checking/shaving again
            yield return new WaitForSeconds(cutInterval);
        }
    }

    private IEnumerator TriggerClipperCut()
    {
        Vector3 cutPosition = clipperBladeTip.position;
        Vector3 cutDirection = clipperBladeTip.up;

        // Convert the selected guard (cm) to meters to match CuttingManager's units. Null when no guard
        // controller is assigned, which preserves old flat-target behavior exactly as before.
        float? guardTargetLengthMeters = clipperGuardController != null
            ? clipperGuardController.CurrentGuardCm / 100f
            : (float?)null;

        if (HasCutAreaBoxes())
        {
            yield return cuttingManager.ExecuteHairCardMeshCutBoxesRoutine(cutAreaBoxes, cutDirection, null, guardTargetLengthMeters);
            yield break;
        }

        cuttingManager.ExecuteHairCardMeshCutCylinder(cutPosition, cutDirection, cutCylinderLength, cutRadius, guardTargetLengthMeters);
    }

    private bool HasCutAreaBoxes()
    {
        if (cutAreaBoxes == null)
            return false;

        for (int i = 0; i < cutAreaBoxes.Length; i++)
        {
            if (cutAreaBoxes[i] != null && cutAreaBoxes[i].enabled)
                return true;
        }

        return false;
    }

    private void OnDrawGizmosSelected()
    {
        Gizmos.color = Color.blue;

        if (HasCutAreaBoxes())
        {
            Matrix4x4 oldMatrix = Gizmos.matrix;

            for (int i = 0; i < cutAreaBoxes.Length; i++)
            {
                BoxCollider box = cutAreaBoxes[i];

                if (box == null || !box.enabled)
                    continue;

                Gizmos.matrix = box.transform.localToWorldMatrix;
                Gizmos.DrawWireCube(box.center, box.size);
            }

            Gizmos.matrix = oldMatrix;
            return;
        }

        if (clipperBladeTip == null)
            return;

        Vector3 cutAxis = clipperBladeTip.up.normalized;
        Vector3 cylinderHalf = cutAxis * (cutCylinderLength * 0.5f);
        Vector3 cylinderStart = clipperBladeTip.position - cylinderHalf;
        Vector3 cylinderEnd = clipperBladeTip.position + cylinderHalf;

        Gizmos.DrawWireSphere(cylinderStart, cutRadius);
        Gizmos.DrawWireSphere(cylinderEnd, cutRadius);

        if (clipperBladeTip != null)
        {
            Gizmos.DrawLine(cylinderStart + clipperBladeTip.right * cutRadius, cylinderEnd + clipperBladeTip.right * cutRadius);
            Gizmos.DrawLine(cylinderStart - clipperBladeTip.right * cutRadius, cylinderEnd - clipperBladeTip.right * cutRadius);
            Gizmos.DrawLine(cylinderStart + clipperBladeTip.forward * cutRadius, cylinderEnd + clipperBladeTip.forward * cutRadius);
            Gizmos.DrawLine(cylinderStart - clipperBladeTip.forward * cutRadius, cylinderEnd - clipperBladeTip.forward * cutRadius);
        }
    }

    void Start()
    {
        hapticIntensity = 3;
        toolID = 3;
        //toolModel = 
        //toolAnimator = 
    }

    public override void useTool()
    {
        ToggleClipper();
    }
    
    public override int getCurrentToolID()
    {
        return toolID; // Updated to match your toolID setting
    }

    public override void playHapticFeedback(float duration)
    {
        // Left empty for your VR controller haptic implementation
    }  
}
